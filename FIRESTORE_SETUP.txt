// Instrucciones para configurar roles en Firestore
// 
// 1. Abre Firebase Console: https://console.firebase.google.com
// 2. Selecciona tu proyecto: experience-app-56e09
// 3. Ve a Firestore Database
// 4. Crea una nueva colección llamada "users"
// 5. Agrega los siguientes documentos:
//
// Documento 1 - ADMIN:
// Collection: users
// Document ID: [UID de admin@gmail.com]
// Contenido:
// {
//   "email": "admin@gmail.com",
//   "role": "admin",
//   "createdAt": "2024-01-01T00:00:00.000Z"
// }
//
// Documento 2 - CUSTOMER:
// Collection: users
// Document ID: [UID de henryperalta@gmail.com]
// Contenido:
// {
//   "email": "henryperalta@gmail.com",
//   "role": "customer",
//   "createdAt": "2024-01-01T00:00:00.000Z"
// }
//
// NOTA: Los UIDs los puedes encontrar en:
// Firebase Console > Authentication > Users
//
// Alternativamente, la app creará automáticamente estos documentos
// en el primer login de cada usuario con role "customer" por defecto.
// Luego puedes cambiar el rol desde Firebase Console.
